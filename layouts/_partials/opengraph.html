{{- $title := .Title | default .Site.Title -}} {{- $description := .Description | default .Summary | plainify | htmlUnescape -}} {{- $description = $description | replaceRE `\s+` " " | strings.TrimSpace | truncate 160 -}} {{- $description = $description | default $title -}} {{- $image := "" -}} {{- $imageAlt := "" -}} {{- $pageURL := urls.Parse .Permalink -}} {{/* Inspect rendered content so Markdown and shortcode images use the same URLs as the page. */}} {{- range findRE `(?is)]*>` .Content -}} {{- if not $image -}} {{- $src := findRESubmatch `(?is)\ssrc\s*=\s*(?:"([^"]+)"|'([^']+)')` . 1 -}} {{- with $src -}} {{- $match := index . 0 -}} {{- $source := index $match 1 | default (index $match 2) | htmlUnescape -}} {{- $url := $pageURL.ResolveReference (urls.Parse $source) -}} {{- if in (slice "http" "https") $url.Scheme -}} {{- $image = $url.String -}} {{- end -}} {{- end -}} {{- if $image -}} {{- with findRESubmatch `(?is)\salt\s*=\s*(?:"([^"]*)"|'([^']*)')` . 1 -}} {{- $match := index . 0 -}} {{- $imageAlt = index $match 1 | default (index $match 2) | htmlUnescape -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{ if $image -}} {{ with $imageAlt -}} {{ end -}} {{ else -}} {{ end -}}
